# typography.txt — настройки типографики для filter_bgen.lua
# separator=: — разделитель полей
# # — комментарий
#
# Формат:
#   key:value         — [key]{.t} → value (одиночная, класс .t)
#   key:open:close    — [текст]{.key} → open + текст + close (парная)
#

separator=:

# ============================================================
# Одиночные символы (класс .t)
# ============================================================

# --- Тире и многоточие ---
# (smart уже даёт —, – и … но если нужно явно)
mdash:—
ndash:–
minus:−
hellip:…

# --- Неразрывные символы ---
# (Неразрывный пробел указывается в md как "\ ", но можно и так)
nbsp: 
nbh:‑

# --- Градусы и математика ---
deg:°
plusmn:±
times:×
divide:÷
micro:µ

# --- Валюты ---
rub:₽
euro:€
pound:£
yen:¥
cent:¢

# --- Знаки ---
sect:§
para:¶
copy:©
reg:®
trade:™

# --- Пунктуация ---
# (обычно набираются с клавиатуры, но для единообразия)
laquo:«
raquo:»
ldquo:“
rdquo:”
lsquo:‘
rsquo:’

# ============================================================
# Парные символы (класс = ключ)
# ============================================================

# --- Кавычки (русская типографика) ---
q:«:»
qq:„:“
qqq:‘:’

# --- Кавычки (английские, если понадобятся) ---
en:”:”

# --- Угловые скобки (typographic angle brackets, не математические < >) ---
angle:⟨:⟩
